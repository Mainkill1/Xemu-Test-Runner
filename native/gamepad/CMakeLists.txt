cmake_minimum_required(VERSION 3.20)
project(xemu_gamepad LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(xemu-gamepad worker.cpp pad-device.cpp)
target_link_libraries(xemu-gamepad PRIVATE Threads::Threads)
add_executable(gamepad-core-checks ../../tests/gamepad/check_core.cpp)
target_include_directories(gamepad-core-checks PRIVATE .)
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "Windows gamepad adapter requires MSVC and Windows SDK C++/WinRT, not a custom driver")
  endif()
  target_compile_definitions(xemu-gamepad PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(xemu-gamepad PRIVATE windowsapp)
endif()
foreach(target xemu-gamepad gamepad-core-checks)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME gamepad-core COMMAND gamepad-core-checks)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME gamepad-fixture-heartbeat
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/gamepad/check_fixture_heartbeat.py")
  set_tests_properties(gamepad-fixture-heartbeat PROPERTIES TIMEOUT 15)
endif()
install(TARGETS xemu-gamepad RUNTIME DESTINATION tools)
if(UNIX)
  install(FILES linux-sdl-mapping.txt DESTINATION tools)
endif()
