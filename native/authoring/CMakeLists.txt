cmake_minimum_required(VERSION 3.20)
project(xemu_authoring_media LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.24 gstreamer-sdp-1.0 gstreamer-webrtc-1.0 json-glib-1.0)
add_executable(xemu-authoring-media stream-worker.c)
target_compile_definitions(xemu-authoring-media PRIVATE GST_USE_UNSTABLE_API)
target_link_libraries(xemu-authoring-media PRIVATE PkgConfig::GST)
if(WIN32)
  target_link_libraries(xemu-authoring-media PRIVATE user32)
else()
  find_package(X11 REQUIRED)
  target_link_libraries(xemu-authoring-media PRIVATE X11::X11)
endif()
if(MSVC)
  target_compile_options(xemu-authoring-media PRIVATE /W4)
else()
  target_compile_options(xemu-authoring-media PRIVATE -Wall -Wextra -Werror)
endif()
install(TARGETS xemu-authoring-media RUNTIME DESTINATION tools)
